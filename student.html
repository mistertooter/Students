/* ========================================================== STUDENT GRADE TRACKING DATABASE ========================================================== */ /* ========================================================== 1. STUDENTS ========================================================== */ CREATE TABLE Students ( Student_ID INT PRIMARY KEY, First_Name VARCHAR(50) NOT NULL, Last_Name VARCHAR(50) NOT NULL, Address VARCHAR(100), City VARCHAR(50), County VARCHAR(50), Age INT ); /* ========================================================== 2. TEACHERS ========================================================== */ CREATE TABLE Teachers ( Teacher_ID INT PRIMARY KEY, First_Name VARCHAR(50) NOT NULL, Last_Name VARCHAR(50) NOT NULL, Department VARCHAR(100) ); /* ========================================================== 3. COURSES ========================================================== */ CREATE TABLE Courses ( Course_ID VARCHAR(20) PRIMARY KEY, Course_Name VARCHAR(100) NOT NULL, Teacher_ID INT NOT NULL, Credits DECIMAL(3,1) NOT NULL, FOREIGN KEY (Teacher_ID) REFERENCES Teachers(Teacher_ID) ); /* ========================================================== 4. ENROLLMENT Connects students to courses. ========================================================== */ CREATE TABLE Enrollments ( Enrollment_ID INT PRIMARY KEY, Student_ID INT NOT NULL, Course_ID VARCHAR(20) NOT NULL, Semester VARCHAR(20) NOT NULL, School_Year INT NOT NULL, FOREIGN KEY (Student_ID) REFERENCES Students(Student_ID), FOREIGN KEY (Course_ID) REFERENCES Courses(Course_ID) ); /* ========================================================== 5. ATTENDANCE ========================================================== */ CREATE TABLE Attendance ( Attendance_ID INT PRIMARY KEY, Enrollment_ID INT NOT NULL, Class_Date DATE NOT NULL, Status VARCHAR(10) NOT NULL, FOREIGN KEY (Enrollment_ID) REFERENCES Enrollments(Enrollment_ID) ); /* ========================================================== 6. ASSIGNMENTS ========================================================== */ CREATE TABLE Assignments ( Assignment_ID INT PRIMARY KEY, Enrollment_ID INT NOT NULL, Assignment_Name VARCHAR(100), Score DECIMAL(5,2), Maximum_Score DECIMAL(5,2), FOREIGN KEY (Enrollment_ID) REFERENCES Enrollments(Enrollment_ID) ); /* ========================================================== 7. QUIZZES ========================================================== */ CREATE TABLE Quizzes ( Quiz_ID INT PRIMARY KEY, Enrollment_ID INT NOT NULL, Quiz_Name VARCHAR(100), Score DECIMAL(5,2), Maximum_Score DECIMAL(5,2), FOREIGN KEY (Enrollment_ID) REFERENCES Enrollments(Enrollment_ID) ); /* ========================================================== 8. TESTS ========================================================== */ CREATE TABLE Tests ( Test_ID INT PRIMARY KEY, Enrollment_ID INT NOT NULL, Test_Name VARCHAR(100), Score DECIMAL(5,2), Maximum_Score DECIMAL(5,2), FOREIGN KEY (Enrollment_ID) REFERENCES Enrollments(Enrollment_ID) ); /* ========================================================== STUDENT GRADE REPORT ========================================================== */ SELECT S.Student_ID AS "Student ID", S.First_Name AS "First Name", S.Last_Name AS "Last Name", S.Address, S.City, S.County, S.Age, C.Course_ID AS "Course ID", C.Course_Name AS "Course", CONCAT( T.First_Name, ' ', T.Last_Name ) AS "Teacher", E.Semester, E.School_Year AS "Year", C.Credits, /* Assignment Average */ ( SELECT ROUND( AVG((A.Score / NULLIF(A.Maximum_Score,0)) * 100), 2 ) FROM Assignments A WHERE A.Enrollment_ID = E.Enrollment_ID ) AS "Assignment Average", /* Quiz Average */ ( SELECT ROUND( AVG((Q.Score / NULLIF(Q.Maximum_Score,0)) * 100), 2 ) FROM Quizzes Q WHERE Q.Enrollment_ID = E.Enrollment_ID ) AS "Quiz Average", /* Test Average */ ( SELECT ROUND( AVG((X.Score / NULLIF(X.Maximum_Score,0)) * 100), 2 ) FROM Tests X WHERE X.Enrollment_ID = E.Enrollment_ID ) AS "Test Average" FROM Students S JOIN Enrollments E ON S.Student_ID = E.Student_ID JOIN Courses C ON E.Course_ID = C.Course_ID JOIN Teachers T ON C.Teacher_ID = T.Teacher_ID ORDER BY S.Last_Name, S.First_Name;
